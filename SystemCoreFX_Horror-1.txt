local pulses = {}

local function startPulse(name, peak, attack, duration)
    pulses[name] = {
        peak = peak,
        t0 = os.clock(),
        a = math.max(attack, 0.01),
        d = math.max(duration, 0.01)
    }
end

local function samplePulse(name, now)
    local p = pulses[name]
    if not p then
        return 0
    end
    local e = (now or os.clock()) - p.t0
    if e >= p.a + p.d then
        pulses[name] = nil
        return 0
    end
    if e < p.a then
        return p.peak * (e / p.a)
    end
    local k = 1 - (e - p.a) / p.d
    return p.peak * k * k
end

local function newLayerSound(name, speed, volume, looped, parent)
    local s = Instance.new("Sound")

local function playSegment(snd, dur, speed, volume)
    if not snd or not snd.Parent then
        return
    end

    local token = (snd:GetAttribute("Token") or 0) + 1
    snd:SetAttribute("Token", token)

    pcall(function()
        snd.PlaybackSpeed = speed or 1
        if volume then
            snd.Volume = volume
        end
        if not snd:GetAttribute("OwnAsset") and snd.TimeLength > 1 then
            snd.TimePosition = math.random() * math.max(0, snd.TimeLength - dur * (speed or 1) - 0.5)
        else
            snd.TimePosition = 0
        end
        snd:Play()
    end)

    local s = session
    task.delay(math.max(0.2, dur - 0.3), function()
        if s ~= session or not snd.Parent or snd:GetAttribute("Token") ~= token then
            return
        end
        TweenService:Create(snd, TweenInfo.new(0.3), {Volume = 0}):Play()
        task.delay(0.35, function()
            if snd.Parent and snd:GetAttribute("Token") == token then
                snd:Stop()
            end
        end)
    end)
end

local function createGlow(adornee, name, studs, bases, color, alwaysOnTop)

local function toRel(cf)
    local p = cf.Position
    return CFrame.new(p * SCALE) * (CFrame.new(-p) * cf)
end

local function addMonsterPart(model, name, size, relCF, color, material, group)
    local part = createPart(model, name, size * SCALE, color, material or Enum.Material.Neon)
    part.CastShadow = false
    local rel = toRel(relCF)
    monsterRel[part] = rel
    monsterGroup[part] = group or "body"
    table.insert(monsterParts, part)
    monsterBaseCFrames[part] = monsterBaseRoot * rel
    part.CFrame = monsterBaseRoot * rel
    return part
end

local function addMonsterLimb(model, name, thick, depth, len, startCF, jointRot, color, group)

    addMonsterPart(model, name, Vector3.new(thick, len, depth), cf * CFrame.new(0, -len / 2, 0), color, Enum.Material.Neon, group)
    return cf * CFrame.new(0, -len, 0)
end

local function getRandomMonsterPosition()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if not root then
        local g = Vector3.new(0, 5, 0)
        return g + Vector3.new(0, FEET_DROP * SCALE, 0), g
    end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    local filter = {character}
    if fxFolder then
        table.insert(filter, fxFolder)
    end
    rayParams.FilterDescendantsInstances = filter

    local ground
    local fallbackGround

    for _ = 1, 6 do
        local angle = math.random() * math.pi * 2
        local distance = math.random(MONSTER_MIN_DISTANCE, MONSTER_MAX_DISTANCE)

        local x = math.cos(angle) * distance
        local z = math.sin(angle) * distance

        local rayOrigin = root.Position + Vector3.new(x, 600, z)
        local rayDirection = Vector3.new(0, -1200, 0)

        local result = Workspace:Raycast(rayOrigin, rayDirection, rayParams)

        if result then
            ground = result.Position
            break
        end

        if not fallbackGround then
            fallbackGround = Vector3.new(root.Position.X + x, root.Position.Y - 3, root.Position.Z + z)
        end
    end

    ground = ground or fallbackGround
    return ground + Vector3.new(0, FEET_DROP * SCALE, 0), ground
end

local function isExcludedPart(obj)
    if obj:IsA("Terrain") then
        return true
    end

    if not obj.Anchored then
        return true
    end

    if fxFolder and obj:IsDescendantOf(fxFolder) then
        return true
    end

    if monster and obj:IsDescendantOf(monster) then
        return true
    end

    local a = obj.Parent
    while a and a ~= Workspace do
        if a:IsA("Model") and a:FindFirstChildOfClass("Humanoid") then
            return true
        end
        a = a.Parent
    end

    return false
end

local function updateFloor(now)
    local n = #floorParts
    if n == 0 then
        return
    end

    local seqPos = (now * FLOOR_COLOR_SPEED) % #colorSequence
    local i0 = math.floor(seqPos)
    local frac = seqPos - i0
    local c1 = colorSequence[i0 + 1]
    local c2 = colorSequence[(i0 + 1) % #colorSequence + 1]
    local targetColor = c1:Lerp(c2, frac)

    local hot = samplePulse("floorHot", now)
    if hot > 0 then
        targetColor = targetColor:Lerp(Color3.fromRGB(255, 0, 0), math.min(hot, 1))
    end

    local shakePulse = samplePulse("floorShake", now)
    local jitterMul = 0.35 + shakePulse * 0.65
    local rippleAmp = 0.15 + shakePulse * 1.2 + hot * 0.5

    local center = monsterCurrentGround
    local count = math.min(n, FLOOR_BUDGET_PER_FRAME)

    for _ = 1, count do
        local obj = floorParts[floorCursor]
        floorCursor = floorCursor % n + 1

        if obj and obj.Parent then
            obj.Color = targetColor

            local origCF = originalCFrames[obj]

            if origCF then
                local offsetX = math.random(-350, 350) / 100 * jitterMul
                local offsetZ = math.random(-350, 350) / 100 * jitterMul
                local offsetY = math.random(-150, 150) / 100 * jitterMul

                if center then
                    local p = origCF.Position
                    local dx = p.X - center.X
                    local dz = p.Z - center.Z
                    local dist = math.sqrt(dx * dx + dz * dz)
                    offsetY = offsetY + math.sin(dist * 0.045 - now * 5) * rippleAmp
                end

                obj.CFrame = origCF + Vector3.new(
                    offsetX,
                    offsetY,
                    offsetZ
                )
            end
        end
    end
end

local function updateLighting(now, dt)
    if not lightingActive then
        return
    end

    atmosphereLevel = math.min(1, atmosphereLevel + dt / LIGHTING_RAMP_TIME)
    local level = atmosphereLevel * atmosphereLevel * (3 - 2 * atmosphereLevel)

    local dim = math.min(samplePulse("blackout", now), 1)
    local flash = samplePulse("redFlash", now)
    local fogThick = samplePulse("fogThick", now)
    local reveal = samplePulse("reveal", now)
    local roarL = samplePulse("roarLight", now)

    local breathe = 0.5 + 0.5 * math.sin(now * 0.7)
    local faulty = clamp(math.noise(now * 0.8, 3) * 1.2, -0.3, 0.3)
    local pulseMul = clamp(0.75 + 0.2 * breathe + faulty, 0.45, 1.05)

    local keep = 1 - 0.85 * dim

    Lighting.Brightness = lerp(originalLighting.Brightness, (0.35 + 0.5 * flash + 0.2 * roarL) * keep, level)

    local amb = Color3.fromRGB(clamp((80 * pulseMul + 140 * flash + 40 * roarL) * keep, 0, 255), 0, 0)
    local outAmb = Color3.fromRGB(clamp((55 * pulseMul + 100 * flash + 30 * roarL) * keep, 0, 255), 0, 0)
    Lighting.Ambient = originalLighting.Ambient:Lerp(amb, level)
    Lighting.OutdoorAmbient = originalLighting.OutdoorAmbient:Lerp(outAmb, level)
    Lighting.ColorShift_Top = originalLighting.ColorShift_Top:Lerp(Color3.fromRGB(clamp(130 * pulseMul + 80 * flash, 0, 255), 0, 0), level)
    Lighting.ColorShift_Bottom = originalLighting.ColorShift_Bottom:Lerp(Color3.fromRGB(70 * pulseMul, 0, 0), level)

    Lighting.FogColor = originalLighting.FogColor:Lerp(Color3.fromRGB(clamp(70 + 60 * flash, 0, 255), 0, 0), level)
    Lighting.FogStart = lerp(originalLighting.FogStart, 60, level)

    local fogEndTarget = 1500 - fogThick * 800 + reveal * 500
    local origEnd = math.max(originalLighting.FogEnd, 1)
    Lighting.FogEnd = math.exp(lerp(math.log(origEnd), math.log(fogEndTarget), level))

    Lighting.ExposureCompensation = originalLighting.ExposureCompensation + (-0.25 * level) + 0.3 * flash - 0.35 * dim

    if atmosphereObj and atmosphereObj.Parent then
        atmosphereObj.Density = lerp(originalAtmosphere.Density, 0.38 + 0.18 * fogThick, level)
        atmosphereObj.Offset = lerp(originalAtmosphere.Offset, 0.25, level)
        atmosphereObj.Color = originalAtmosphere.Color:Lerp(Color3.fromRGB(150 + 50 * flash, 10, 10), level)
        atmosphereObj.Decay = originalAtmosphere.Decay:Lerp(Color3.fromRGB(190, 0, 0), level)
        atmosphereObj.Glare = lerp(originalAtmosphere.Glare, 0, level)
        atmosphereObj.Haze = lerp(originalAtmosphere.Haze, 2.2 + fogThick * 1.5, level)
    end

    if colorCorrection and colorCorrection.Parent then
        colorCorrection.TintColor = Color3.new(1, 1, 1):Lerp(Color3.fromRGB(255, 150, 150), level)
        colorCorrection.Saturation = -0.15 * level
        colorCorrection.Contrast = 0.1 * level
        colorCorrection.Brightness = -0.03 * level - 0.08 * dim + 0.05 * flash
    end
end

local function restoreLighting()
    if not lightingActive then
        return
    end

    lightingActive = false

    Lighting.Brightness = originalLighting.Brightness
    Lighting.Ambient = originalLighting.Ambient
    Lighting.OutdoorAmbient = originalLighting.OutdoorAmbient
    Lighting.ColorShift_Top = originalLighting.ColorShift_Top
    Lighting.ColorShift_Bottom = originalLighting.ColorShift_Bottom
    Lighting.FogColor = originalLighting.FogColor
    Lighting.FogStart = originalLighting.FogStart
    Lighting.FogEnd = originalLighting.FogEnd
    Lighting.ExposureCompensation = originalLighting.ExposureCompensation

    if atmosphereObj then
        if atmosphereCreated then
            atmosphereObj:Destroy()
        elseif atmosphereObj.Parent then
            atmosphereObj.Density = originalAtmosphere.Density
            atmosphereObj.Offset = originalAtmosphere.Offset
            atmosphereObj.Color = originalAtmosphere.Color
            atmosphereObj.Decay = originalAtmosphere.Decay
            atmosphereObj.Glare = originalAtmosphere.Glare
            atmosphereObj.Haze = originalAtmosphere.Haze
        end
    end
    atmosphereObj = nil
    atmosphereCreated = false

    if colorCorrection then
        colorCorrection:Destroy()
        colorCorrection = nil
    end

    atmosphereLevel = 0
end

local function updateAudio(now, dt)
    local k = math.min(1, dt * 5)
    local lvl = atmosphereLevel
    local windSwell = samplePulse("windSwell", now)
    local rumbleSwell = samplePulse("rumbleSwell", now)
    local quake = math.min(samplePulse("quake", now), 1.5)
    local breathe = 0.5 + 0.5 * math.sin(now * 0.5)
    local gust = clamp(0.6 + math.noise(now * 0.35, 11) * 0.8, 0.15, 1)

    local function setVol(s, target)
        if s then
            s.Volume = s.Volume + (target - s.Volume) * k
        end
    end

    setVol(layerSounds.Rumble, (0.5 + 1.6 * rumbleSwell + 0.6 * quake) * lvl)
end

local function playerYawRelativeToMonster()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not (root and monsterWorldPosition) then
        return 0
    end
    local rel = (CFrame.new(monsterWorldPosition) * monsterFacing):PointToObjectSpace(root.Position)
    return math.atan2(-rel.X, -rel.Z)
end

local behaviorList = {}
for name, def in pairs(behaviorDefs) do
    table.insert(behaviorList, {name = name, w = def.w})
end

local function weightedPick(list, filterFn)
    local total = 0
    for _, item in ipairs(list) do
        if not filterFn or filterFn(item) then
            total = total + item.w
        end
    end

    if total <= 0 then
        return nil
    end

    local r = math.random() * total
    for _, item in ipairs(list) do
        if not filterFn or filterFn(item) then
            r = r - item.w
            if r <= 0 then
                return item
            end
        end
    end
    return nil
end

local function startBehavior(name, now)
    local def = behaviorDefs[name]
    if not def then
        return
    end

    for key in pairs(bTgt) do
        bTgt[key] = 0
    end

    def.apply(bTgt)
    behavior.name = name
    behavior.untilT = now + def.dur()
    lastBehavior = name
end

local function stepBehavior(now)
    if now < behavior.untilT then
        return
    end

    if behavior.name == "idle" then
        local picked = weightedPick(behaviorList, function(item)
            return item.name ~= lastBehavior
        end)
        if picked then
            startBehavior(picked.name, now)
        end
    else
        for key in pairs(bTgt) do
            bTgt[key] = 0
        end
        behavior.name = "idle"
        behavior.untilT = now + rand(2.5, 5.5)
    end
end

local function makeArmCF(s, raise, phase, now)
    local pivot = Vector3.new(s * 13, 66, 0) * SCALE
    local ang = raise * 1.1 + math.sin(now * 0.9 + phase) * 0.03
    return CFrame.new(pivot) * CFrame.Angles(ang, 0, s * (raise * 0.35 + 0.02)) * CFrame.new(-pivot)
end

local function updateMonster(now, dt)
    if not (monster and monster.Parent and monsterWorldPosition) then
        return
    end

    stepBehavior(now)

    local alpha = 1 - math.exp(-dt * 1.3)
    for key, v in pairs(bCur) do
        bCur[key] = v + (bTgt[key] - v) * alpha
    end

    if now >= nextBlinkTime then
        startPulse("eyeBlink", 1, 0.04, 0.4)
        nextBlinkTime = now + rand(2.5, 7)
    end

    local tremble = bCur.tremble
    local shake = math.sin(now * 5) * 0.8 * SCALE * (0.35 + tremble)
        + math.noise(now * 20, 3) * 2 * SCALE * tremble
    local float = math.sin(now * 1.7) * 2 * SCALE * 0.7
    local rotation = math.sin(now * 1.2) * 0.035
    local sway = math.sin(now * 0.6) * 0.02

    local stomp = samplePulse("stomp", now)
    local reveal = samplePulse("reveal", now)
    local flare = samplePulse("eyeFlare", now)
    local blink = samplePulse("eyeBlink", now)
    local roarLight = samplePulse("roarLight", now)

    local approachOffset = Vector3.new(0, 0, 0)
    if reveal > 0 then
        approachOffset = approachDir * monsterApproachMax * math.min(reveal, 1)
    end

    local lift = bCur.lift - stomp * 4 * SCALE
    local rootPos = monsterWorldPosition + approachOffset + Vector3.new(0, float + lift, 0)
    local rootCF = CFrame.new(rootPos)
        * monsterFacing
        * CFrame.Angles(0, rotation + bCur.bodyYaw, sway)
        * CFrame.new(shake, 0, 0)

    monsterCurrentGround = monsterGroundPosition + approachOffset

    local headPivot = Vector3.new(0, 84, -2) * SCALE
    local headT = CFrame.new(headPivot)
        * CFrame.Angles(bCur.headPitch, bCur.headYaw, bCur.headRoll)
        * CFrame.new(-headPivot)

    local jawOpen = clamp(bCur.jaw + samplePulse("jaw", now) * 0.45 + 0.03 * (math.sin(now * 1.1) * 0.5 + 0.5), 0, 0.6)
    local jawPivot = Vector3.new(0, 87, -2) * SCALE
    local jawT = headT
        * CFrame.new(jawPivot)
        * CFrame.Angles(-jawOpen, 0, 0)
        * CFrame.new(-jawPivot)

    groupCF.head = headT
    groupCF.jaw = jawT
    groupCF.armL = makeArmCF(-1, bCur.armL, 0, now)
    groupCF.armR = makeArmCF(1, bCur.armR, 1.7, now)

    for _, part in ipairs(monsterParts) do
        if part and part.Parent then
            local rel = monsterRel[part]

            if rel then
                part.CFrame = rootCF * groupCF[monsterGroup[part]] * rel
            end
        end
    end

    local baseEye = 0.65 + 0.2 * math.sin(now * 2.3)
    local eyeLevel = clamp(baseEye + bCur.eyeBoost * 0.3 + flare * 0.9 - blink * 1.1, 0, 1.6)
    local white = clamp((eyeLevel - 0.9) * 1.4, 0, 1)
    local eyeColor = Color3.fromRGB(255, 20 + 170 * white, 20 + 150 * white)

    for _, e in ipairs(monsterEyes) do
        e.Color = eyeColor
        e.Transparency = clamp(blink, 0, 1) * 0.85
    end

    if monsterMouthGlow then
        monsterMouthGlow.Color = Color3.fromRGB(150 + 105 * clamp(jawOpen * 2 + roarLight, 0, 1), 0, 0)
    end

    if monsterLight then
        monsterLight.Brightness = (12 + math.abs(math.sin(now * 4)) * 22)
            * (1 + flare * 0.8 + bCur.auraBoost * 0.6 + roarLight * 0.8)
    end

    if monsterHighlight then
        monsterHighlight.FillTransparency = clamp(0.8 - reveal * 0.25 - flare * 0.1 - bCur.auraBoost * 0.08, 0.45, 0.9)
        monsterHighlight.OutlineTransparency = clamp(0.35 - reveal * 0.2, 0.05, 0.6)
    end

    glowFrameCounter = glowFrameCounter + 1
    if glowFrameCounter % 2 == 0 then
        local auraLevel = clamp(0.45 + 0.3 * (0.5 + 0.5 * math.sin(now * 0.8)) + 0.8 * flare + bCur.auraBoost * 0.5 + roarLight * 0.6 + reveal * 0.5, 0, 1.5)
        local eyeGlowLevel = clamp(eyeLevel, 0, 1.5)

        for _, g in ipairs(monsterGlow) do
            local lvl = g.kind == "eye" and eyeGlowLevel or auraLevel
            for i, f in ipairs(g.frames) do
                f.BackgroundTransparency = clamp(1 - (1 - g.base[i]) * lvl, 0, 1)
            end
        end
    end

    if monsterShadow then
        local sizeMul = 1 + math.max(lift, 0) * 0.003
        monsterShadow.Size = Vector3.new(0.4, 230 * sizeMul, 140 * sizeMul)

        local off = monsterFacing:VectorToWorldSpace(Vector3.new(
            sway * 900 + shake * 0.5 + bCur.headRoll * 30,
            0,
            -(bCur.armL + bCur.armR) * 25
        ))

        monsterShadow.CFrame = CFrame.new(monsterCurrentGround + off + Vector3.new(0, 0.4, 0))
            * monsterFacing
            * CFrame.Angles(0, 0, math.rad(90))
        monsterShadow.Transparency = clamp(0.55 + math.max(lift, 0) * 0.003, 0.4, 0.8)
    end

    if monsterFogPart then
        monsterFogPart.CFrame = CFrame.new(monsterCurrentGround + Vector3.new(0, 12, 0))
    end
end

local function eventBlackout()
    local s = session
    task.spawn(function()
        local flickers = math.random(2, 3)
        for _ = 1, flickers do
            if s ~= session or not isActive then
                return
            end
            startPulse("blackout", rand(0.4, 0.75), 0.06, rand(0.25, 0.45))
            playSegment(layerSounds.Zap, 0.25, 2.2, 0.5)
            task.wait(rand(0.45, 0.8))
        end

        if s ~= session or not isActive then
            return
        end

        if math.random() < 0.5 then
            startPulse("blackout", 0.9, 0.08, 1.3)
            playSegment(layerSounds.Thump, 1.4, 0.22, 1.2)
        end
    end)
end

local function eventQuake()
    local p = weightedPick(quakePresets)
    if not p then
        return
    end

    startPulse("quake", p.amp, p.att, p.dur)
    startPulse("floorShake", math.min(1, p.amp / 1.2), p.att, p.dur)
    startPulse("rumbleSwell", 1, p.att, p.dur + 1)

    if p.amp >= 0.9 then
        playSegment(layerSounds.Thump, 1.6, 0.22, 1.8)
        startPulse("vignette", 0.35, 0.4, p.dur + 1)
        startPulse("overlay", 0.18 + p.amp * 0.1, 0.4, p.dur + 1)
    end

    if p.amp >= 1.5 then
        bloodTrace(3)
    end
end

local nextEventTime = 0
local eventLastRun = {}

local function runDirector(now)
    if now < nextEventTime then
        return
    end

    nextEventTime = now + rand(DIRECTOR_MIN_GAP, DIRECTOR_MAX_GAP)

    local picked = weightedPick(events, function(e)
        return now - (eventLastRun[e.name] or -1000) >= e.cd
    end)

    if picked then
        eventLastRun[picked.name] = now
        pcall(picked.run)
    end
end

local renderConnection
local slowAcc = 0

renderConnection = RunService.RenderStepped:Connect(function(dt)
    if not isActive then
        return
    end

    local now = os.clock()

    runDirector(now)
    updateFloor(now)
    updateMonster(now, dt)
    followCamera()

    slowAcc = slowAcc + dt
    if slowAcc >= 0.05 then
        local sdt = slowAcc
        slowAcc = 0
        updateLighting(now, sdt)
        updateOverlay(now)
        updateParticles(now)
        updateAudio(now, sdt)
    end
end)

local function cameraShakeStep()
    if not isActive then
        return
    end

    local camera = Workspace.CurrentCamera

    if camera then
        local t = os.clock()
        local strength = earthquakeStrength + samplePulse("quake", t)

        local earthquakeX = (math.sin(t * earthquakeSpeed) * 0.6 + math.noise(t * 7, 0.5) * 1.6) * strength
        local earthquakeY = (math.sin(t * (earthquakeSpeed * 1.37)) * 0.6 + math.noise(t * 7, 1.5) * 1.6) * strength
        local earthquakeZ = (math.cos(t * (earthquakeSpeed * 0.83)) * 0.6 + math.noise(t * 7, 2.5) * 1.6) * strength

        camera.CFrame =
            camera.CFrame
            * CFrame.new(
                earthquakeX,
                earthquakeY,
                earthquakeZ
            )
            * CFrame.Angles(
                earthquakeY * 0.025,
                earthquakeX * 0.025,
                earthquakeZ * 0.025
            )
    end
end

pcall(function()
    RunService:UnbindFromRenderStep("SystemCoreCameraShake")
end)

local function activateEffects()
    session = session + 1
    pulses = {}
    resetBehaviorState()
    eventLastRun = {}
    nextEventTime = os.clock() + 4
    slowAcc = 0

    createWorldFX()
    updateFloorList()
    activateLighting()
    createMonster()
    createOverlayFX()
    startAudio()

    RunService:BindToRenderStep("SystemCoreCameraShake", Enum.RenderPriority.Camera.Value + 1, cameraShakeStep)
end

local function deactivateEffects()
    session = session + 1
    pulses = {}

    pcall(function()
        RunService:UnbindFromRenderStep("SystemCoreCameraShake")
    end)

    restoreOriginalState()
    restoreLighting()
    destroyMonster()
    destroyWorldFX()
    destroyOverlayFX()
    stopAudio()
end

toggleButton.MouseButton1Click:Connect(function()
    if dragMoved then
        dragMoved = false
        return
    end

    isActive = not isActive
    updateButtonColor()

    if isActive then
        activateEffects()
    else
        deactivateEffects()
    end
end)

local isShutdown = false

local function shutdown()
    if isShutdown then
        return
    end
    isShutdown = true

    isActive = false

    if renderConnection then
        renderConnection:Disconnect()
    end

    for _, c in ipairs(connections) do
        pcall(function()
            c:Disconnect()
        end)
    end

    deactivateEffects()

    pcall(function()
        sound:Destroy()
    end)

    pcall(function()
        audioFolder:Destroy()
    end)

    destroyOverlayFX()
end

screenGui.AncestryChanged:Connect(function()
    if not screenGui.Parent then
        shutdown()
    end
end)

    pcall(function()

    end)
end
